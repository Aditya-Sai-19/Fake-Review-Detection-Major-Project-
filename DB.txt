create database amazonreviews;
use amazonreviews;

create table register(retailer_name varchar(50),
gender varchar(20),
contact_no  varchar(12),
address varchar(60),
email varchar(40),
username varchar(40) primary key,
password varchar(40));

create table submit_reviews(reviewer_name varchar(50),
review_text varchar(200),
ratings int,
review_date date);
